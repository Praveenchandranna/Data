pipelines:
  custom: # Pipelines that can only be triggered manually
    Validation_QA:
      - variables:
          - name: Feature_Branch
          - name: Salesforce_Test_Run
            default: RunSpecifiedTests  
            allowed-values:         
             - RunSpecifiedTests
             - RunLocalTests
             - RunAllTestsInOrg
             - NoTestRun
      - step:
          name: Checkout Branch
          image: atlassian/default-image:2
          script:
            - git checkout $Feature_Branch
      - step:
          name: Install Dependencies
          image: node:14
          caches:
            - node
          script:
            - npm install sfdx-cli --global
            - npm install
          
      - step:
          name: Display Branch Name and Test Class Content
          script:
            - echo "Branch name $Feature_Branch"
            - file_content=$(cat ./manifest/"$Feature_Branch"_TestClass.txt)
            - echo "File content:"
            - echo "$file_content"

      - step:
          name: Generate JWT Key
          script:
            - echo "$SALESFORCE_JWT_SECRET_KEY" > server.key

      - step:
          name: Authenticate with Salesforce Org and Validation
          image: salesforce/salesforcedx:latest
          script:
            - echo "Branch name $Feature_Branch"
            - file_content=$(cat ./manifest/"$Feature_Branch"_TestClass.txt)
            - echo "File content:"
            - echo "$file_content"
            - sfdx force:auth:jwt:grant --clientid $SFDC_CLIENT_ID_QA --jwtkeyfile ./JWT/server.key --username $SFDC_USERNAME_QA --setdefaultdevhubusername --setalias QA
            - echo "Authentication completed successfully."
            - |
              if [[ "${Salesforce_Test_Run}" == "RunSpecifiedTests" ]]; then
                test_run="RunSpecifiedTests"
                sfdx force:source:deploy --checkonly -w 500 -x ./manifest/"$Feature_Branch"_package.xml -u $SFDC_USERNAME_QA --testlevel "$Salesforce_Test_Run" --runtests "$file_content"
                echo "Validation completed successfully with RunSpecifiedTests."
              elif [[ "${Salesforce_Test_Run}" == "NoTestRun" ]]; then
                test_run="NoTestRun"
                sfdx force:source:deploy --checkonly -w 500 -x ./manifest/"$Feature_Branch"_package.xml -u $SFDC_USERNAME_QA --testlevel "$Salesforce_Test_Run"
                echo "Validation completed successfully with NoTestRun."
              elif [[ "${Salesforce_Test_Run}" == "RunLocalTests" ]]; then
                test_run="RunLocalTests"
                sfdx force:source:deploy --checkonly -w 500 -x ./manifest/"$Feature_Branch"_package.xml -u $SFDC_USERNAME_QA --testlevel "$Salesforce_Test_Run"
                echo "Validation completed successfully with RunLocalTests."
              elif [[ "${Salesforce_Test_Run}" == "RunAllTestsInOrg" ]]; then
                test_run="RunAllTestsInOrg"
                sfdx force:source:deploy --checkonly -w 500 -x ./manifest/"$Feature_Branch_package".xml -u $SFDC_USERNAME_QA --testlevel "$Salesforce_Test_Run"
                echo "Validation completed successfully with RunAllTestsInOrg."
              fi
